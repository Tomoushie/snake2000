// ---- GetMetricsSnapshot() ----
        public GPUMetricsSnapshot GetMetricsSnapshot()
        {
            if (_disposed) throw new ObjectDisposedException(GetType().Name);
            lock (_metricsLock)
            {
                return new GPUMetricsSnapshot(_lastMetrics);
            }
        }
